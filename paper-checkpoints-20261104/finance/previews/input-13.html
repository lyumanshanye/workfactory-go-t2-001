1function irr_lambda = irrational_lambda( lambda, lambda_reference, lag_years, parameters )
2% USAGE: irrational_lambda( lambda, lambda_reference, lag_years, parameters )
3%
4% lambda and lambda_reference can be both vectors of the same length. The diagnostic belief is using
5% lambda_reference as the reference point, and then distor the distribution on lambda.
6% lag_years specifies the look-back years for diagnostic expectation.
7
8theta_diagnostic = parameters.theta_diagnostic;
9lambda_H = parameters.lambda_H;
10lambda_L = parameters.lambda_L;
11
12a = parameters.lambdaHL;
13b = parameters.lambdaLH;
14aH = (b+a*exp(-(a+b)*lag_years)) / (a+b);
15bH = 1-aH;
16aL = (b-b*exp(-(a+b)*lag_years)) / (a+b);
17bL = 1 - aL;
18
19lambdaT_lambdaL = aH*(lambda_reference-lambda_L) + aL*(lambda_H-lambda_reference);
20lambdaH_lambdaT = bH*(lambda_reference-lambda_L) + bL*(lambda_H-lambda_reference);
21ratio= (lambdaT_lambdaL./lambdaH_lambdaT) ./ ((lambda-lambda_L)./(lambda_H-lambda)) ;
22irr_lambda = (lambda-lambda_L) .* (lambda_H-lambda_L)./( ratio.^theta_diagnostic.*(lambda_H-lambda) + (lambda-lambda_L) ) + lambda_L;
