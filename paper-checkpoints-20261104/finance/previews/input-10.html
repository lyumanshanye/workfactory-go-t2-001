1function par = initialization( par_input )
2% Key Points: if par_input is not given, then initialize the parameters as a rational model, par.model_name has three values: 'benchmark', 'rational', and 'behavioral'.
3% If par_input is given, then we should use the crisis frequency from par_input.
4
5if(nargin<1) % No input. Then just set the model as a rational model, and crisis target as the 12.2%
6 par.model_name = 'rational'; % default is the rational model
7 % (0) Credit spread
8 par.crisis_frequency_target = 0.122;
9 par.bond_default_probability = 0.04;
10 par.loss_given_default = 0.55; % Loss given default is 55%
11 par.hat_kappa0 = par.loss_given_default - 0.06; % note: the 6% is the average price decline of capital during a crisis. This one will be adjusted after the model is solved.
12 par.tau = 7; % 7 years
13 par.pi = par.bond_default_probability/(par.crisis_frequency_target);
14
15 par.alg_control.plot = false; par.alg_control.show_progress=false; par.alg_control.solve_everything_again=false;
16
17 % (1) Key parameters related to bank run and beliefs
18 alpha = 0.05; beta=1;
19 % Parameter trials: alpha=0.1, beta=0.1 is too low that results in negligible declines in productivity.
20 lambda_L=0.001; lambda_H = 0.5;
21 lambdaHL=0.5;
22 lambdaLH = fzero( @(lambdaLH) lambdaLH / (lambdaLH+lambdaHL) * lambda_H + lambdaHL / (lambdaLH+lambdaHL) * lambda_L - par.crisis_frequency_target, 0.15 );
23
24 % diagnostic belief parameters
25 theta_diagnostic = 0.9;
26 lag_years = 1;
27
28 % (2) Other parameters
29 AL=0.12; AH=0.15;
30 rho = 0.04; sigmaK=0.04; delta=0.1; chi=3;
31 eta=0.03;
32 e = 0.001; varepsilon=0.00001; % Bank Run parameters.
33 par.HQ_Cutoff=0.03;
34else
35 par = par_input;
36 if( ~isfield(par, 'loss_given_default' ) )
37 par.loss_given_default = 0.55;
38 end
39 % (1) Key parameters related to bank run and beliefs
40 alpha = par_input.alpha; beta=par_input.beta;
41 % Parameter trials: alpha=0.1, beta=0.1 is too low that results in negligible declines in productivity.
42 lambda_L=par_input.lambda_L; lambda_H = par_input.lambda_H;
43 lambdaHL= par_input.lambdaHL; lambdaLH=par_input.lambdaLH;
44 % diagnostic belief parameters
45 theta_diagnostic = par_input.theta_diagnostic;
46 lag_years = par_input.lag_years;
47
48 % (2) Other parameters
49 AL=par_input.AL; AH=par_input.AH;
50 rho = par_input.rho; sigmaK=par_input.sigmaK; delta=par_input.delta; chi=par_input.chi;
51 eta=par_input.eta;
52 e = par_input.e; varepsilon=par_input.varepsilon; % Bank Run parameters.
53 par.HQ_Cutoff = par_input.HQ_Cutoff;
54
55 if(isfield(par_input, 'model_name'))
56 par.model_name = par_input.model_name;
57 else % If there is not a model name field, then treat this model as a rational model.
58 par.model_name = 'rational';
59 end
60 % if( strncmp(par_input.model_name, 'benchmark',20) ) % if this is for the benchmark model without belief variation, set the belief as the average belief in the rational case.
61 % epsilon = 0.001;
62 % lambda_L = par_input.lambda_H*(1-epsilon);
63 % lambda_H = par_input.lambda_H;
64 % lambdaHL= 0.5; lambdaLH=0.1; % note: the underlying intensity is always the same. However, lambda_H is what drives the frequency of the liquidity shock. This facilitates model calibration.
65 % end
66 par.pi = par_input.bond_default_probability/(par_input.tau*par_input.crisis_frequency_target); % Update the pi according to the default probability in par_input.
67end
68
69% (3) Functions
70muK_fun = @(p) (p-1)/chi + delta; % This is the amount of actual "investment"
71investment = @(p) (p-1).^2/(2*chi) + (p-1)./chi + delta ; % This is the investment adjustment cost, or the real resources consumed.
72investmentD = @(p) p/chi ;
73% Jump
74kappa_lambda_fun = @(lambda) (lambda - lambda_L).*(lambda_H-lambda)./lambda; % Jump in lambda
75mu_lambda_fun = @(lambda) (lambda_L - lambda).*lambdaHL + (lambda_H - lambda).*lambdaLH - (lambda-lambda_L).*(lambda_H-lambda); % Drift in lambda
76
77% (3) Limit values
78p0 = fsolve( @(p) investment(p) + rho*p - AL, 1 , optimoptions( 'fsolve','Display','none' ) ); % The initial condition
79p1 = fsolve( @(p) investment(p) + rho*p - AH, 1 , optimoptions( 'fsolve','Display','none' ) ); % The end value at w=1, or psi=1
80
81% (4) Record parameters for call-back inside functions. par is a struct that includes all the relevant parameters for other functions
82par.AH = AH; par.AL = AL; par.rho = rho; par.sigmaK = sigmaK;
83par.max_jump = 0.5; par.delta = delta; par.eta=eta;
84par.alpha = alpha; par.beta=beta;
85par.lambda_H = lambda_H; par.lambda_L = lambda_L; par.lambdaHL=lambdaHL; par.lambdaLH=lambdaLH;
86par.theta_diagnostic = theta_diagnostic; par.lag_years = lag_years;
87par.kappa_lambda_fun =kappa_lambda_fun; par.mu_lambda_fun = mu_lambda_fun;
88par.investment=investment; par.investmentD=investmentD;
89par.muK_fun = muK_fun;
90par.chi = chi;
91par.p0 = p0; par.p1=p1;
92par.e = e; par.varepsilon = varepsilon;
93
94% (4) Grid
95N_w = 60;
96N_lambda = 10;
97lambda_grid = [ exp( linspace( log(lambda_L+0.001), log( max(lambda_H/5,lambda_L*0.9+0.1*lambda_H) ) , 5 )')-0.001 ; linspace( max(lambda_H/5*2,lambda_L*0.9+0.1*lambda_H), lambda_H, N_lambda-5 )' ];
98w_left = [ 0 ; exp( linspace(log(0.001), log(0.4), round(N_w/4*3) )' ) ] ;
99w_star = max(w_left); w_rest = linspace( w_star+ w_left(end)-w_left(end-1) , 1, N_w - numel(w_left) )';
100w_grid = [w_left; w_rest ];
101w_matrix = kron( w_grid, ones( 1, N_lambda ) );
102lambda_matrix = kron( ones( N_w, 1 ), lambda_grid' );
103
104% (6) Record the grid and algorithm control information
105par.w_grid = w_grid; par.lambda_grid = lambda_grid;
106par.N_w = N_w; par.N_lambda = N_lambda;
107par.w_matrix = w_matrix; par.lambda_matrix = lambda_matrix;
108dt = 1/12; par.dt = dt; % The time interval for simulation
109par.tol = 0.8*10^(-3); % The tolerance for iteration errors in the 2D case.
110par.print_level = 0;
