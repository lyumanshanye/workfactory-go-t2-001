37# === Generate Differences ======
38GenDiff <- function( variable, Lag=1 ){
39 index = 1:(length(variable)-abs(Lag))
40 if(Lag>0){
41 return( c( variable[index+Lag] - variable[index], rep(NA,Lag)) )
42 }else{
43 Lag = -Lag
44 return( c(rep(NA,Lag), variable[index+Lag] - variable[index]) )
45 }
46}
47GenLogDiff <- function( variable, Lag=1 ){ # Diff negative means shifting forward.
48 return(GenDiff(log(variable),Lag))
49}
50# ==== Gen Lagged Variables ======
51Lag_fun <- function(x, lag=1){
52 # The default is the lag of variables. If lag<0, then it becomes the forward of variables.
53 if(lag>0){
54 return( c( rep(NA,lag), x[1:(length(x)-lag)] ) )
55 }else{
56 lag = abs(lag)
57 return( c( x[(lag+1):length(x)], rep(NA,lag) ) )
58 }
59}

79# === Robust Standard Errors ===
80RobustSE <- function( regs, lag_input=12, error_type="NeweyWest" ){
81 robust_se.list = list()
82 for(i in 1:length(regs)){
83 if(error_type=="NeweyWest"){
84 if(lag_input==1){
85 results = unclass(coeftest(regs[[i]]))
86 }else{
87 results = unclass(coeftest(regs[[i]], vcov. = NeweyWest, lag=lag_input))
88 }
89 }else{
90 # just typical robust standard errors
91 results = unclass(coeftest(regs[[i]], vcov. = vcovHC(regs[[i]], type = 'HC0')))
92 }
93 robust_se.list[[i]] <- results[,2]
94 }
95 return( robust_se.list )
96}

117# === Remove outliers ====
118process_outliers <- function(x, provide_index=FALSE, lambda=0.5, return_remaining_index=FALSE){
119 x <- ts( x, start= c(1991,1) , frequency=365)
120 result <- tsoutliers(x, iterate = 0, lambda = lambda)
121 if(provide_index){
122 print( paste0(length(result$index), " outliers detected!" ) )
123 if(return_remaining_index){
124 return(setdiff( seq(1,length(x)) , result$index))
125 }else{
126 return(result$index)
127 }
128 }else{
129 x[result$index] = result$replacements
130 print( paste0(length(result$index), " outliers detected and processed!" ) )
131 return( as.numeric(x) )
132 }
133}

135# ====== Residualize Results ======
136residualize <- function(x, reference){
137 index = !is.na(x) & !is.na(reference)
138 reg = lm( x[index] ~ reference[index] )
139 residual = x - reg$coefficients[2]*reference
140 return( residual - min(residual,na.rm=TRUE) )
141}

143# === Numerical Derivative ===
144derivative_fun <- function(x_vec, y_vec){
145 N = length(y_vec)
146 dy = y_vec*0
147 dy[2:N] = (y_vec[2:N]-y_vec[1:(N-1)])/(x_vec[2:N]-x_vec[1:(N-1)])
148 dy[1] = (y_vec[2]-y_vec[1])/(x_vec[2]-x_vec[1])
149 return(dy)

152# === Function for Data Smoothing.===
153library(fda)
154data_smoothing <- function( x, y, smooth_order=4 ){
155 N = length(x)
156 nbasis = N + smooth_order - 2
157 wbasis <- create.bspline.basis( range(x), nbasis, smooth_order)
158
159 cvec0 <- matrix(0,nbasis,1)
160 Wfd0 <- fd(cvec0, wbasis)
161
162 # set up functional parameter object
163 Lfdobj <- 2 # penalize curvature of acceleration
164 lambda <- 10^(-0.5) # smoothing parameter
165 growfdPar <- fdPar(Wfd0, Lfdobj, lambda)
166
167 weights = c( 1000, rep(1, length(x)-1) )
168
169 result = tryCatch( smooth.monotone(x, y, growfdPar, weights), error=function(e){ return("Error") } )
170
171 if( class(result)!="monfd" ){ # if the monotonic smoother cannot work, that normally is because too many fluctuations
172 fitted_values = lowess(x, y)$y
173 fitted_values[1] = y[1]
174 }else{
175 beta = result$beta
176 fitted_values = beta[1] + beta[2]*eval.monfd(x, result$Wfdobj)
177 }
178 return( fitted_values )
179}

182# defining transparent colors
183t_col <- function(color, transparency_percent = 50, name = NULL) {
184 # color = color name
185 # percent = % transparency
186 # name = an optional name for the color
187 ## Get RGB values for named color
188 gen_color <- function(color_ind){
189 rgb.val <- col2rgb(color_ind)
190 ## Make new color using input color as base and alpha set by transparency
191 t.col <- rgb(rgb.val[1], rgb.val[2], rgb.val[3], maxColorValue = 255,
192 alpha = (100 - transparency_percent) * 255 / 100,
193 names = name)
194 ## Save the color
195 return(t.col)
196 }
197 if(length(color)==1){
198 return( gen_color(color) )
199 }else{
200 colors_vec = rep(gen_color(color[1]),length(length(color)))
201 for(i in 1:length(color)){
202 colors_vec[i] = gen_color(color[i])
203 }
204 return(colors_vec)
205 }
206}

221 # Note: x is a vector, and yseries_input is either a dataframe or a vector
222 if(sum(is.na(colors))>0){
223 colors = rep(c( "blue", "red", "purple", "azure4", "darkslategray4", "goldenrod2", "cyan" ), 2)
224 }
225 if(sum(is.na(line_types))>0){
226 line_types = rep( seq(1,6), 2 )
227 }
228 yseries_input = as.data.frame(yseries_input)
229 N.dim = ncol( yseries_input )
230 if( (is.null(x) > 0) ){ # If there is no input from x and only one column is supplied from yseries_input
231 if( ncol(yseries_input)==1 ){

234 x = yseries_input[,1]

238 DT = cbind( x, yseries_input )
239 DT = as.data.frame(DT)
240 NumNAs = apply( DT, 1, function(z) sum(is.na(z))) # Eliminate NAs
241 Num_Non_NAs = apply( DT, 1, function(z) sum(!is.na(z)))

247 DT = DT[ order(DT$x), ] # Order the table by x

265 if( N.dim>2 ){ # More than 1 series
266 for( i in 2:(N.dim-1) ){
267 lines(DT[,1] , DT[,i+1], col=colors[i], lwd=Linewidth_vec[i], lty=line_types[i] )
268 }
269 }
270 if(!is.null(standard_errors)){
271 colors_shaded = t_col(colors, transparency_percent = CI_transparency)
272 standard_errors = as.data.frame(standard_errors)

285 }else{
286 legend( LegendPosition , legend = LegendNames, bty = "n", lty= line_types[seq(1, N.dim-1)] , col=colors[seq(1, N.dim-1)], lwd=Linewidth*rep(1,N.dim-1) )
