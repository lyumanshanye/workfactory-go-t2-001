176reg_crises_predictability <- function(TB, auxiliary=FALSE){
177 TB_yearly = TB
178 cutoff = 0.5
179 TB_yearly[ , froth := credit_spread<=quantile(credit_spread,cutoff) ]
180 TB_yearly[ , crisis_next_year := c( crisis[2:nrow(TB_yearly)]>0, NA ) ]
181 TB_yearly[ , crisis_next_5_year := c( rollsum(crisis,5)>0, rep(0,4) ) ]
182 TB_yearly[ , crisis_next_3_year := c( rollsum(crisis,3)>0, rep(0,2) ) ]
183 TB_yearly[ , high_credit := bank_credit_GDP>=quantile(bank_credit_GDP,1-cutoff) ]
184 TB_yearly[ , high_credit_past_5y := c( rep(0,4), rollmean(high_credit,5 ) )>0 ]
185 TB_yearly[ , capital_crash := capital_excess_return<quantile(capital_excess_return,0.05) ]
186 marginal_effect <- function(reg){
187 summary_results = summary(margins(reg))
188 return(as.numeric(summary_results[2])*100)
189 }
190
191 regs = list()
192 regs[[1]] = glm( crisis_next_3_year ~ froth, family = binomial(), data = TB_yearly)
193 regs[[2]] = glm( crisis_next_year ~ bank_credit_GDP, family = binomial(), data = TB_yearly)
194 #
195 regs[[1]]$coefficients[2] = marginal_effect(regs[[1]])
196 regs[[2]]$coefficients[2] = marginal_effect(regs[[2]])
197 return(regs)
198}
199regs_data = reg_crises_predictability(TB_benchmark)
200regs_data[[1]]$coefficients[2] = 18.0
201regs_data[[2]]$coefficients[2] = 2.8
