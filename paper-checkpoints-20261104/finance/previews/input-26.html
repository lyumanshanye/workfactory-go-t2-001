1function results = solve_for_states_from_cs_and_lvg( current_model , leverage, cs, m_simulation )
2
3disp( [ '----------------- Solving the initial states for the ', current_model.par.model_name, ' model --------------------' ] )
4
5if(nargin<4)
6 m_simulation = simulation_results_output(current_model);

9par = current_model.par;
10% Find the correct scaling for the credit spread.
11w_dim = kron(ones(par.N_lambda,1), par.w_grid );
12lambda_dim = kron(par.lambda_grid, ones(par.N_w,1));
13fit_a_function = @(values_matrix) scatteredInterpolant(w_dim, lambda_dim , reshape( values_matrix, par.N_w*par.N_lambda, 1 ), 'linear') ;
14spread_fitted = fit_a_function(current_model.credit_spread);
15avg_spread_before_normalization = mean( spread_fitted(m_simulation.w,m_simulation.lambda) ) ;
16spread_fitted = fit_a_function(current_model.credit_spread/avg_spread_before_normalization); % Re-adjust with the average
17lvg_fitted = fit_a_function( current_model.xK_matrix );
18
19solve_w_and_lambda_fun = @( states ) [ lvg_fitted( states(1), states(2) ) - leverage; spread_fitted( states(1), states(2) ) - cs ] ; % states = [w, lambda]
20opt_solver = optimset('Display','on');
21[ solution, e ] = fsolve( solve_w_and_lambda_fun, [ 0.08, (par.lambda_L+par.lambda_H)/2 ] , opt_solver );
22w = solution(1); results.w=w;
23lambda = solution(2); % This is the "believed" lambda.
24
25lambda= max(min(lambda,par.lambda_H), par.lambda_L);
26if( lambda<=par.lambda_L || lambda>=par.lambda_H || sum(abs(e))>10^(-5) )
27 disp( ['Error in solving the appropriate lambda! lambda=',num2str(lambda)] );
28else
29 disp( ['Initial w=', num2str(w) ,', Initial lambda=', num2str(lambda) ] )
30end
31
32if( strcmp(par.model_name, 'behavioral' ) )
33 % Solve for the current rational lambda for the behavioral model
34 lambda_believed = lambda;
35 residual_underlying_lambda = @(lambda_underlying) irrational_lambda( lambda_underlying, par.lambda_reference, par.lag_years, par) - lambda_believed;
36 opt_solver = optimset('Display','off');
37 [ solution, ~ ] = fsolve( residual_underlying_lambda, lambda_believed , opt_solver );
38 lambda_underlying = solution(1);
39 results.lambda = lambda_underlying;
40 results.lambda_reference = par.lambda_reference;
41 results.lambda_believed = lambda;
42 disp( ['Underlying lambda for behavioral belief is ', num2str(lambda_underlying) ] )
43else
44 results.lambda = lambda;
45end
