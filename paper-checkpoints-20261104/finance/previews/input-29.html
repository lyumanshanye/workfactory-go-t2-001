51%% -------------- Simulate the General-Belief Model (Appendix) Under Different Values of theta_belief and lambda_mid --------------------------------------------------
52disp('----------------------------------------------------------------------------------------------------------')
53disp('--------------------- Simulated Data under General Beliefs ---------------------')
54sol_behavioral = load( ['solutions/baseline_behavioral.mat' ] );
55sol_behavioral = post_processing(sol_behavioral.model_sol);
56par = sol_behavioral.par;
57moments = simulation_results_output( sol_behavioral, 5000 );
58avg_lambda = mean(moments.lambda_rational); % needed for the definitions of beliefs below.
59
60% Plot the functional forms of each case.
61belief_fun1 = @(lambda) max( par.lambda_L + 0.001*lambda, min( par.lambda_H - 0.001*lambda, lambda + 1.2 * sqrt( lambda - par.lambda_L ) .* ( par.lambda_H - lambda ) ) );
62belief_fun2 = @(lambda) max( par.lambda_L + 0.001*lambda, min( par.lambda_H - 0.001*lambda, 3 * ( lambda - par.lambda_L ).^3 ) );
63belief_fun3 = @(lambda) max( lambda, min( par.lambda_H - 0.001*lambda, lambda + 2.3* sqrt( lambda - par.lambda_L ) .* ( 2*avg_lambda - lambda ) ) );
64belief_fun4 = @(lambda) (lambda>avg_lambda).*min( par.lambda_H-0.001*par.lambda_H + 0.001*lambda ,lambda + 1* ( lambda - avg_lambda ).^(1/3) .* ( par.lambda_H - lambda )) + (lambda<=avg_lambda).* lambda ;
65belief_fun5 = @(lambda) min(lambda, 97* ( lambda - par.lambda_L ).^4) ;
66belief_fun6 = @(lambda) min( lambda, min( par.lambda_H - 0.001*lambda, avg_lambda + 4.5 * ( lambda - avg_lambda ).^3 ) );
67
68N_years = 20000;
69belies_list = { belief_fun1, belief_fun2, belief_fun3, belief_fun4, belief_fun5, belief_fun6};

71for(iter = 1:length(belies_list))
72 sol = sol_behavioral;

74 rng(1); % set seed so that it is replicable.
