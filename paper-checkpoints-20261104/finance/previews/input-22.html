33gen_risk_premium_measures <- function(TB, sigmaK){
34 TB[, risk_premium:=lambda*(kappap+alpha)/(1-xK*kappap-alpha*(xK-1)) + (vol_of_p+sigmaK)^2*xK]
35 TB[, total_risk_unadjusted:=lambda*(kappap+alpha)^2 + (vol_of_p+sigmaK)^2 ]
36 TB[, total_bank_equity_risks:=(total_risk_unadjusted) * xK^2 ]
37 TB[,dw_quadratic_var:=lambda*(xK*kappap+alpha*pmax(xK-1,0))^2 + (vol_of_p+sigmaK)^2*xK^2 ]
38 TB[,total_Sharpe:=risk_premium/sqrt(total_risk_unadjusted)] # This is the total Sharpe ratio.
39 return(TB)
40}

45 gen_non_clustered_crisis <- function( crisis_vec ){
46 diff_vec = diff(which(crisis_vec==1))
47 crisis_sparse = crisis_vec
48 crisis_last = which(crisis_vec)[1]
49 for( index in which(crisis_vec)[2:length(diff_vec)] ){
50 if(index-crisis_last<0*12){ # 3 years of non consecutive crises
51 crisis_sparse[index] = 0
52 }else{
53 crisis_sparse[index] = 1
54 crisis_last = index
55 }
56 }
57 return(crisis_sparse)
58 }
59 credit_monthly_growth = c(diff(TB$bank_credit_GDP), 0) / TB$bank_credit_GDP
60 equity_monthly_change = TB$equity_monthly_change
61 solve_for_cutoff <- function(cutoff){
62 if(crisis_def == "bank_credit_GDP_change"){
63 crisis_vec = credit_monthly_growth < cutoff
64 }else{ # equity crisis
65 crisis_vec = equity_monthly_change < cutoff
66 }
67 crisis_sparse = gen_non_clustered_crisis(crisis_vec)
68 DT = data.table( Year=TB$Year, crisis_sparse )
69 crisis_sparse_yearly = DT[ , list(crisis_sparse=mean(crisis_sparse)>0), by=Year ]$crisis_sparse
70 return( mean(crisis_sparse_yearly) - 0.04 )
71 }
72 if(is.na(cutoff)){
73 result = uniroot(solve_for_cutoff, lower=quantile(credit_monthly_growth,0.001), upper=quantile(credit_monthly_growth,0.5) )
74 cutoff=result$root
75 }
76 if(crisis_def == "bank_credit_GDP_change"){
77 print( paste0("Crisis cut off for bank credit growth is ", round(cutoff,digits=2) ) )
78 TB[ , crisis := gen_non_clustered_crisis(credit_monthly_growth < cutoff) ]
79 }else{ # equity crisis
80 print( paste0("Crisis cut off for bank equity growth is ", round(cutoff,digits=2) ) )
81 TB[ , crisis := gen_non_clustered_crisis(equity_monthly_change < cutoff) ]
82 }
83 return(TB)

89gen_yearly_data <- function(TB){
90 TB[ , Year:= floor(T) ]
91 TB[ , p_yearly_return:= c( rep(NA,12), p[13:nrow(TB)]/p[1:(nrow(TB)-12)] - 1 ) ]
92 TB[ ,p_vol_empirical := c(NA, roll_sd( p[2:nrow(TB)]/p[1:(nrow(TB)-1)] - 1 , width=12)) ]
93 TB[is.na(p_vol_empirical) ,p_vol_empirical := mean_fun(p_vol_empirical) ] # For the initial 12 NA observations, just replace them with average volatility.
94 TB[ ,p_vol_empirical := p_vol_empirical/mean_fun(p_vol_empirical) ]
95 TB[ , K_monthly_growth := c(rep(0,1), logK_vec_original[2:nrow(TB)]-logK_vec_original[1:(nrow(TB)-1)] ) ]
96 TB[ , K_monthly_growth := winsorize(K_monthly_growth, lower_quantile=0.001, upper_quantile=0.999) ]
97 TB[, capital_return_monthly := c( rep(NA,1) , K_monthly_growth[2:nrow(TB)] + log(p[2:nrow(TB)]/p[1:(nrow(TB)-1)]) ) ]
98 # Define the monthly return volatilty, in yearly unit (i.e., think about the continuous-time limit of the volatility on sample path)
99 TB[, capital_return_vol_simulation := c( rep(NA,12) , 12*roll_sd( capital_return_monthly[13:nrow(TB)] , width=12) ) ]
100 TB[is.na(capital_return_vol_simulation) , capital_return_vol_simulation := mean_fun(capital_return_vol_simulation) ]
101 TB_qunatities = TB[ , list( Year, dN, crisis, total_bank_equity ) ]
102 TB_yearly_quantities = TB_qunatities[, lapply(.SD, mean,na.rm=TRUE), by=Year]
103 # deal with some outliers due to numerical issues
104 TB[rK_monthly_from_sim>quantile(TB$rK_monthly_from_sim,0.999,na.rm=TRUE)]$rK_monthly_from_sim = NA
105 TB[,rK_yearly:= exp(rollsum( log(1+rK_monthly_from_sim), 12, fill=NA, align="right")) - 1 ]
106 TB_yearly_prices = TB[ , list( Year, w, xK, lambda, lambda_rational, underlying_lambda, K, GDP, productivity, bank_equity, vol_of_p, dB_premia, bank_credit,
107 p, credit_spread, rf, rK, rd, equity_excess_return, sharpe_ratio, investment_rate, capital_excess_return, p_vol_empirical, kappap, p_yearly_return,
108 bank_credit_GDP, rK_yearly, logK_vec_original, total_Sharpe, risk_premium, total_risk_unadjusted, capital_return_vol_simulation) ]
109 # Note: the variable "investment rate" maps to phi(mu^K) in the model, which is the cost of investment, or the real resources cost by investment.
110 TB_yearly_prices = TB_yearly_prices[ , lapply(.SD, first), by=Year ]
111 TB_yearly = merge(TB_yearly_quantities, TB_yearly_prices, by="Year")
112 TB_yearly[ , crisis:=crisis>0 ]
113 TB_yearly[ , dN:=dN>0 ]
114 return( TB_yearly[ Year>=1 ] )
115}

117processing_simulation_data <- function(TB_yearly){
118 # Note: the input table should be at yearly frequency
119 TB_yearly[ , bank_credit_GDP_pre_year := c( mean(bank_credit_GDP), bank_credit_GDP[1:(nrow(TB_yearly)-1)] ) ]
120 TB_yearly[, bank_equity_excess_return:= equity_excess_return]
121 TB_yearly[ , sigma_of_p_standardized := vol_of_p/mean(vol_of_p) ]
122 TB_yearly[ , total_vol_standardized := sqrt(total_risk_unadjusted) / mean_fun(sqrt(total_risk_unadjusted)) ] # Standardize the theory-implied total volatility
123 weights = 0.8^c(10:0)
124 TB_yearly[ , historical_vol_p := roll_sd(p, 10, weights ) ]
125 TB_yearly[ is.na(historical_vol_p)]$historical_vol_p = mean(TB_yearly$historical_vol_p,na.rm=TRUE)
126 TB_yearly[ , Kp := K*p ]
127 TB_yearly[ , capital_growth := c( K[2:nrow(TB_yearly)]/K[1:(nrow(TB_yearly)-1)]-1, 0 ) ]
128 TB_yearly[ , investment_rate_3Y:= c( rollsum( investment_rate, 3 ), rep( 3*mean(TB_yearly$investment_rate) ,2)) ]
129 index = 1:(nrow(TB_yearly)-1)
130 # Next-year capital return
131 # Average credit spread: the average last 5 year value of credit spread, following Krishnamurthy and Muir definition
132 TB_yearly[ , avg_credit_spread_5Y := c( rep(0,5-1), rollmean(credit_spread, k=5) ) ]
133 TB_yearly[, froth:= avg_credit_spread_5Y<median(avg_credit_spread_5Y)] # This is the definition in KM 2017.
134 crises_index = which( TB_yearly$crisis==1 )
135 distress_index = which( TB_yearly$dN==1 )
136 pre_crises_years = 5
137 index_before_crises = crises_index
138 index_before_distress = distress_index
139 for( i in 1:pre_crises_years ){
140 index_before_crises = c( index_before_crises, pmax(crises_index-i,1) )
141 index_before_distress = c( index_before_distress, pmax(distress_index-i,1) )
142 }
143 TB_yearly[, before_crises_1_to_5Y:=0]
144 TB_yearly[ index_before_crises,before_crises_1_to_5Y:=1]
145 TB_yearly[ , before_crises:=0]
146 TB_yearly[ crises_index-1, before_crises:=1]
147 TB_yearly[ , large_credit := bank_credit_GDP>=quantile(bank_credit_GDP,0.9) ]
148 TB_yearly[ , before_crises_and_large_credit := large_credit*before_crises ]
149 TB_yearly[ , credit_growth_past_year := c(rep(0,12), bank_credit[13:nrow(TB_yearly)]/bank_credit[1:(nrow(TB_yearly)-12)]-1) ]
150 TB_yearly[ , credit_growth := c(0, (bank_credit[2:nrow(TB_yearly)]/bank_credit[1:(nrow(TB_yearly)-1)]-1) ) ]
151 TB_yearly[ , credit_growth_next_year := c( bank_credit[2:nrow(TB_yearly)]/bank_credit[1:(nrow(TB_yearly)-1)]-1, 0 ) ]
152 TB_yearly[ , capital_excess_return_normalized := capital_excess_return/mean(capital_excess_return)*0.059 ] # Note: 0.2 is the standard deviation of equity returns.
153 TB_yearly[ , capital_crash := capital_excess_return<quantile(capital_excess_return,0.05) ]
154
155 TB_yearly[ , credit_spread_next_year := c( credit_spread[2:nrow(TB_yearly)] , rep(0,1) ) ]
156 TB_yearly[ , credit_spread_change := c( credit_spread[2:nrow(TB_yearly)] - credit_spread[1:(nrow(TB_yearly)-1)], 0 ) ]
157 TB_yearly[ , credit_spread_last_year := c( rep(0,1), credit_spread[1:(nrow(TB_yearly)-1)] ) ]
158 TB_yearly[ , GDP_yearly_change:= c( GDP[2:nrow(TB_yearly)]/GDP[1:(nrow(TB_yearly)-1)]-1, rep(0,1)) ]
159 TB_yearly[ , GDP_yearly_change_past:= c(rep(0,1), GDP[2:nrow(TB_yearly)]/GDP[1:(nrow(TB_yearly)-1)]-1) ]
160 TB_yearly[ , GDP_3Y_change:= c( GDP[(1+3):nrow(TB_yearly)]/GDP[1:(nrow(TB_yearly)-3)]-1, rep(0,3)) ]
161 TB_yearly[ , GDP_5Y_change:= c( GDP[(1+5):nrow(TB_yearly)]/GDP[1:(nrow(TB_yearly)-5)]-1, rep(0,5)) ]
162
163 # Important: the changes are all forward-looking.
164 TB_yearly[ , credit_2Y_change:= c( bank_credit[(1+2):nrow(TB_yearly)]/bank_credit[1:(nrow(TB_yearly)-2)]-1, rep(0,2)) ]
165 TB_yearly[ , credit_3Y_change:= c( bank_credit[(1+3):nrow(TB_yearly)]/bank_credit[1:(nrow(TB_yearly)-3)]-1, rep(0,3)) ]
166 TB_yearly[ , credit_5Y_change:= c( bank_credit[(1+5):nrow(TB_yearly)]/bank_credit[1:(nrow(TB_yearly)-5)]-1, rep(0,5)) ]
167 TB_yearly[ , credit_8Y_change:= c( bank_credit[(1+8):nrow(TB_yearly)]/bank_credit[1:(nrow(TB_yearly)-8)]-1, rep(0,8)) ]
168 TB_yearly[ , equity_3Y_change:= c( bank_equity[(1+3):nrow(TB_yearly)]/bank_equity[1:(nrow(TB_yearly)-3)]-1, rep(0,3)) ]
169 TB_yearly[ , equity_1Y_change:= c( bank_equity[(1+1):nrow(TB_yearly)]/bank_equity[1:(nrow(TB_yearly)-1)]-1, rep(0,1)) ]
170
171 TB_yearly[ , post_crises_5Y := c( rep(0,4), rollsum(crisis, 5) )>0 ] # An indicate of within 5 years after the next crisis.
172 TB_yearly[ , post_crises_3Y := c( rep(0,2), rollsum(crisis, 3) )>0 ]
173 TB_yearly[ , crises_next_5Y := c( rollsum(crisis, 5), rep(0,4) )>0 ]
174 TB_yearly[ , recession := (GDP_yearly_change<quantile(GDP_yearly_change,0.08)) & (!crises_next_5Y) ]
175 median_GDP_change_in_crisis = median(TB_yearly[crisis==TRUE]$GDP_yearly_change, na.rm=TRUE)
176 TB_yearly[ , GDP_yearly_change_next := c( GDP_yearly_change[2:nrow(TB_yearly)],0 ) ]
177 TB_yearly[ , before_severe_crises := before_crises*(GDP_yearly_change_next<median_GDP_change_in_crisis) ]
178 TB_yearly[ , before_mild_crises := before_crises*(GDP_yearly_change_next>median_GDP_change_in_crisis) ]
179
180 TB_yearly[ , bank_equity_change_around_crisis:=0 ]
181 TB_yearly[ , Years_around_crisis := -100 ] # Check -4 and +6 years range.
182 TB_yearly[ , Years_around_crisis:=as.integer(Years_around_crisis) ]
183 counter_total = sum(TB_yearly$crisis)
184 counter = 0
185 for( index in which(TB_yearly$crisis) ){
186 counter = counter + 1
187 if(counter%%(counter_total/10)==0){
188 print( paste0("Processing years around crisis:", round(counter/counter_total*100,digits = 2), "%" ) )
189 }
190 index_for_change = max(1,index-4):min(index+8,nrow(TB_yearly))
191 index_for_change = index_for_change[ TB_yearly[index_for_change,]$Years_around_crisis==-100 ]
192 if( is.element(0, index_for_change - index) ){ # At least having the starting point of the crisis
193 TB_yearly[index_for_change, Years_around_crisis := index_for_change - index ]
194 TB_yearly[index_for_change,]$bank_equity_change_around_crisis = log(TB_yearly$total_bank_equity[index_for_change]) - log(TB_yearly$total_bank_equity[index])
195 }
196 }
197 TB_yearly[ Years_around_crisis==-100 , Years_around_crisis:= NA ]
198 TB_yearly[ , pre_crisis := Years_around_crisis<0 ]
199 TB_yearly[ is.na(pre_crisis), pre_crisis:=FALSE ]
200 TB_yearly[ , crisis_next_year := c( crisis[2:nrow(TB_yearly)], rep(FALSE,1) ) ]
201
202 # Then investment and consumption growth
203 TB_yearly[,consumption:= (productivity - investment_rate)*K ]
204 TB_yearly[,consumption_growth := c(0, consumption[2:nrow(TB_yearly)]/consumption[1:(nrow(TB_yearly)-1)]-1 ) ]
205 TB_yearly[,investment:=investment_rate*K]
206 TB_yearly[,investment_growth:= c(0, investment[2:nrow(TB_yearly)]/investment[1:(nrow(TB_yearly)-1)]-1 )]
207 return(TB_yearly)
208}
