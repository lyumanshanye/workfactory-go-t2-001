20% % Calculate the initial state of lambda
21solve_w_and_lambda_fun = @( states ) [ lvg_fitted( states(1), states(2) ) - leverage; spread_fitted( states(1), states(2) ) - cs ] ; % states = [w, lambda]

29 disp( ['Initial w=', num2str(w) ,', Initial believed lambda=', num2str(lambda_belief) ] )
30end

32if( strcmp( current_model.par.model_name, 'behavioral' ) )
33 lambda_underlying_fun = @(lambda_underlying) irrational_lambda( lambda_underlying, par.lambda_reference, par.lag_years , par) - lambda_belief;
