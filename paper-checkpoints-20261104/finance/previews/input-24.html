1function sim = simulate( w_init, lambda_init, dB_vec, dN_vec, N_years, model_sol, par )
2% Usage: impulse_response_new( w, lambda, dB, dN, eta_adjustment, N_years, model_sol, par )
3% dB and dN should be of the same length.
4% Notes: (w, lambda) is the starting state to calculate the impulse response.
5% dB_vec and dN_vec store the shocks along different periods. They can be of different length.
6% All other parameters are stored in par, including the simulation time interval dt.
7% w_init=0.1; lambda_init=0.1; dB_vec=zeros(1200, 1 ); dN_vec=zeros(1200, 1 ); N_years=100;, model_sol=model_sol_new; par=model_sol_new.par;
8
9if( par.print_level>0 )
10 disp( '->-> Simulate the Model ->->' );
11end
12if(isfield(par,'scenario'))
13 disp( ['Simulating general belief: ', par.scenario, ' belief'] )
14end
15
16%% Parameters
17N_w = par.N_w; N_lambda = par.N_lambda;
18w_grid = par.w_grid; lambda_grid=par.lambda_grid;
19w_matrix = par.w_matrix;
20delta = par.delta; sigmaK=par.sigmaK;
21dt = par.dt; % simulate the model at dt year interval.
22w_lower = 10^(-3); % to avoid simulation error.
23w_upper = 1-10^(-3);
24
25N_sim = round(N_years/dt);
26T_vec = linspace( 1, N_sim, N_sim )'*dt;
27
28w_dim = kron(ones(N_lambda,1), w_grid(2:(N_w-1)) );
29lambda_dim = kron(lambda_grid, ones(N_w-2,1));
30fit_a_function = @(values_matrix) scatteredInterpolant(w_dim, lambda_dim , reshape( values_matrix(2:(N_w-1),:) , (N_w-2)*N_lambda, 1 ), 'linear', 'linear') ; % note: linear extrapolation at the boundary
31
32if( isfield(par, 'eta_adjustment') ) % adjustment for the eta parameter during simulations.
33 eta_adjustment = par.eta_adjustment;
34else
35 eta_adjustment = 0;
36end
37if( isfield(par, 'lag_years') ) % the number of years for
38 lag_years = par.lag_years;
39else
40 lag_years = 0;
41end
42if( ~isfield(par, 'theta_diagnostic') )
43 par.theta_diagnostic = 0;
44end
45
46% process shocks
47dB_vec = dB_vec(:); dN_vec=dN_vec(:); % translate into column vectors
48dB_vec = [ dB_vec; zeros( N_sim-length(dB_vec), 1 ) ];
49dN_vec = [ dN_vec; zeros(N_sim-length(dN_vec), 1 ) ];
50
51% Evolution of lambda
52mu_lambda_fun = par.mu_lambda_fun;
53kappa_lambda_fun = par.kappa_lambda_fun;
54
55% Evolution functions for state variable w
56muw_matrix_adjusted = model_sol.muw_matrix - eta_adjustment*w_matrix ; % Additional adjustment for growth of muw
57muw_fun = fit_a_function( muw_matrix_adjusted);
58sigmaw_fun = fit_a_function(model_sol.sigmaw_matrix);
59kappaw_fun = fit_a_function(model_sol.kappaw_matrix);
60
61% Evolution of capital K
62muK_fun = par.muK_fun; % Note: muK_fun is a function of price, so we need to fit a price function. To calculate the total drift, we need to add the depreciation on capital as well.
63price_fun = fit_a_function(model_sol.p_matrix);
64kappaK = model_sol.kappaK;
65
66% Evolution of bank equity w
67mub_fun = fit_a_function( model_sol.mub_matrix );
68sigmab_fun = fit_a_function( model_sol.sigmab_matrix );
69kappab_fun = fit_a_function( model_sol.kappab_matrix );
70
71%% Rational Evolutions With shocks
72K_init = 1; % Initial capital is normalized to 1
73wb_init = 1; % Initial bank equity normalized to 1. Bank equity measures the amount of equity for a bank that has never transited into households for the whole periods.
74rational_evolution = zeros( N_sim, 4 ); % Evolution after getting dB_or_dN shocks
75rational_evolution(1, :) = [ w_init, lambda_init, K_init, wb_init ] ;
76
77% Note: use log scale to avoid explosion.
78for( t = 1:(N_sim-1) )
79 w = rational_evolution(t,1);
80 lambda_rational = rational_evolution(t,2);
81 K_log = rational_evolution(t,3);
82 muK = muK_fun(price_fun( w, lambda_rational) );
83 wb = rational_evolution(t,4);
84 rational_evolution(t+1, 1) = min( max( w .* (1 + muw_fun(w, lambda_rational)*dt + sigmaw_fun(w, lambda_rational )*dB_vec(t) - kappaw_fun(w,lambda_rational)*dN_vec(t) ), w_lower ), w_upper );
85 rational_evolution(t+1, 2) = lambda_rational + mu_lambda_fun(lambda_rational)*dt + kappa_lambda_fun(lambda_rational)*dN_vec(t) ;
86 rational_evolution(t+1,3) = K_log + ( muK - delta ) * dt + sigmaK*dB_vec(t) - kappaK*dN_vec(t) ; % Note: this is in the log scale.
87 rational_evolution(t+1,4) = max( wb * ( 1 + mub_fun(w,lambda_rational) * dt + sigmab_fun(w,lambda_rational) * dB_vec(t) - kappab_fun(w,lambda_rational)*dN_vec(t) ), w_lower);
88end
89logK_vec_original = rational_evolution(:,3); % value before detrending
90K_vec_log_detrended = rational_evolution(:,3) - linspace( rational_evolution(1,3), rational_evolution(end,3), size(rational_evolution,1) )';
91rational_evolution(:,3) = exp(K_vec_log_detrended);
92% fit_fun1 = fit_a_function(model_sol.capital_risk_premium);
93% capital_risk_premium_vec = fit_fun1( rational_evolution(:,1), rational_evolution(:,2) );
94% fit_fun2 = fit_a_function(model_sol.equity_risk_premium);
95% equity_risk_premium_vec = fit_fun2( rational_evolution(:,1), rational_evolution(:,2) );
96
97% Find the dynamics of capital and bank equity.
98capital_drift_fun = fit_a_function(model_sol.capital_drift_matrix);
99equity_drift_fun = fit_a_function(model_sol.equity_drift_matrix);
100capital_dB_fun = fit_a_function(model_sol.sigmap_matrix+par.sigmaK);
101equity_dB_fun = fit_a_function( (model_sol.sigmap_matrix+par.sigmaK).*model_sol.xK_matrix );
102capital_dN_fun = fit_a_function(model_sol.kappap_matrix);
103equity_dN_fun = fit_a_function(model_sol.kappab_matrix);
104
105w = rational_evolution(:, 1);
106lambda = rational_evolution(:, 2);
107capital_premium_dt_dB_dN = capital_drift_fun(w,lambda)*dt + capital_dB_fun(w,lambda).*dB_vec - capital_dN_fun(w,lambda).*dN_vec;
108equity_premium_dt_dB_dN = equity_drift_fun(w,lambda)*dt + equity_dB_fun(w,lambda).*dB_vec - equity_dN_fun(w,lambda).*dN_vec;
109
110
111%% Behavioral Evolutions With shocks
112if( par.model_name=="behavioral" ) % if it is a behavioral model
113
114kappaw_fitted_tensor = model_sol.kappaw_fitted_tensor;
115kappap_fitted_tensor = model_sol.kappap_fitted_tensor;
116% initialize the behavioral belief.
117if( isfield( par, 'lambda_reference' ) )
118 lambda_reference = par.lambda_reference;
119else
120 lambda_reference = lambda_init;
121end
122lambda_behavioral_init = irrational_lambda( lambda_init, lambda_reference, lag_years, par );
123behavioral_evolution = zeros( N_sim, 4 ); % Evolution after getting dB_or_dN shocks
124behavioral_evolution(1, :) = [ w_init, lambda_behavioral_init, K_init, wb_init ] ;
125rational_lambda_vec = lambda_init * ones( N_sim, 1 );
126for( t = 1:(N_sim-1) )
127 w = behavioral_evolution(t,1);
128 lambda_rational = rational_lambda_vec(t);
129 lambda_behavioral = behavioral_evolution(t,2);
130 periods = round(lag_years/dt);
131 if(t>periods)
132 lambda_reference = rational_lambda_vec( max(t-periods,1) );
133 end
134 K_log = behavioral_evolution(t,3);
135 wb = behavioral_evolution(t,4); %Note: this one is not as useful, since we already have the w dynamics.
136
137 % update w
138 muK = muK_fun(price_fun( w, lambda_behavioral) );
139 kappaw = kappaw_fitted_tensor( w, lambda_rational, lambda_reference );
140% kappaw = kappaw_fun(w,lambda_rational);
141 behavioral_evolution(t+1, 1) = min( max( w .* (1 + muw_fun(w, lambda_behavioral)*dt + sigmaw_fun(w, lambda_behavioral )*dB_vec(t) - kappaw*dN_vec(t) ) , w_lower ), w_upper );
142
143 % update rational and behavioral lambda
144 lambda_current_rational = lambda_rational + mu_lambda_fun(lambda_rational)*dt + kappa_lambda_fun(lambda_rational)*dN_vec(t) ; % the rational lambda for t+1
145 rational_lambda_vec(t+1) = lambda_current_rational; % assign value
146 if(isfield(par,'scenario'))
147 % lambda_reference = (par.lambda_H * par.lambdaLH + par.lambda_L*par.lambdaHL)/(par.lambdaHL+par.lambdaLH);
148 lambda_reference = 0.087; % this is the average true lambda in the baseline Bayesian calibration.
149 % overoptimistic v.s. overpessimistic
150 if(strcmpi(par.scenario,'optimistic'))
151 behavioral_evolution(t+1, 2) = min( irrational_lambda( lambda_current_rational, lambda_reference, lag_years, par ), lambda_current_rational ) ;
152 else % otherwise it is a pessmistic belief.
153 behavioral_evolution(t+1, 2) = max( irrational_lambda( lambda_current_rational, lambda_reference, lag_years, par ), lambda_current_rational ) ;
154 end
155 else
156 % diagnostic belief
157 behavioral_evolution(t+1, 2) = irrational_lambda( lambda_current_rational, lambda_reference, lag_years, par );
158 end
159
160 % update K
161 behavioral_evolution(t+1,3) = K_log + ( muK - delta ) * dt + sigmaK*dB_vec(t) - kappaK*dN_vec(t) ;
162
163 % update wb
164 kappab = kappap_fitted_tensor( w, lambda_rational, lambda_reference );
165 behavioral_evolution(t+1,4) = wb * ( 1 + mub_fun(w,lambda_behavioral) * dt + sigmab_fun(w,lambda_behavioral) * dB_vec(t) - kappab*dN_vec(t) ) ;
166end
167
168logK_vec_behavioral_original = behavioral_evolution(:,3) ;
169K_vec_log_detrended = behavioral_evolution(:,3) - linspace( behavioral_evolution(1,3), behavioral_evolution(end,3), size(behavioral_evolution,1) )';
170behavioral_evolution(:,3) = exp(K_vec_log_detrended);
171% capital_risk_premium_vec = model_sol.capital_risk_premium_fitted_tensor( behavioral_evolution(:,1), rational_lambda_vec, [ mean(rational_lambda_vec)*ones(periods,1); rational_lambda_vec(1:(end-periods)) ] );
172% equity_risk_premium_vec = model_sol.equity_risk_premium_fitted_tensor( behavioral_evolution(:,1), rational_lambda_vec, [ mean(rational_lambda_vec)*ones(periods,1); rational_lambda_vec(1:(end-periods)) ] );
173
174
175w = behavioral_evolution(:, 1);
176lambda = behavioral_evolution(:, 2);
177lambda_ref = [ rational_lambda_vec(1:periods); rational_lambda_vec( 1:end-periods ) ];
178kappap_vec = model_sol.kappap_fitted_tensor(w,rational_lambda_vec,lambda_ref);
179xK_vec = model_sol.xK_fitted_matrix(w,lambda) ;
180kappawb_vec = kappap_vec.*xK_vec + par.alpha*(xK_vec-1);
181capital_premium_dt_dB_dN = capital_drift_fun(w,lambda)*dt + capital_dB_fun(w,lambda).*dB_vec - (par.alpha+kappap_vec).*dN_vec;
182equity_premium_dt_dB_dN = equity_drift_fun(w,lambda)*dt + equity_dB_fun(w,lambda).*dB_vec - kappawb_vec.*dN_vec;
183
184end
185
186
187
188
189%% Store Results
190sim.T_vec = T_vec;
191sim.N_sim = N_sim;
192sim.dB_vec = dB_vec;
193sim.dN_vec = dN_vec;
194
195sim.rational = simulation_other_outputs( rational_evolution, model_sol, par);
196sim.rational.capital_risk_premium_vec = capital_premium_dt_dB_dN;
197sim.rational.equity_risk_premium_vec = equity_premium_dt_dB_dN;
198sim.rational.lambda_rational = rational_evolution(:,2);
199sim.rational.logK_vec_original = logK_vec_original; % This is the original K_vec without detrending.
200if( par.model_name=="behavioral" ) % if it is the behavioral model, then store the behavioral results.
201 sim.behavioral = simulation_other_outputs( behavioral_evolution, model_sol, par);
202 sim.behavioral.capital_risk_premium_vec = capital_premium_dt_dB_dN;
203 sim.behavioral.equity_risk_premium_vec = equity_premium_dt_dB_dN;
204 sim.behavioral.lambda_rational = rational_lambda_vec;
205 sim.behavioral.logK_vec_original = logK_vec_behavioral_original; % This is the original K_vec without detrending
206end
207
208% Store to the same field name regardless of rational or behavioral
209if( par.model_name~="behavioral" )
210 sim.results = sim.rational;
211else
212 sim.results = sim.behavioral;
213end
