def occupation_rows(source=DEFAULT_SOURCE):
    with (source / "tables/occupation_data.csv").open(encoding="utf-8-sig", newline="") as f:
        return {r["O*NET-SOC Code"]: r for r in csv.DictReader(f)}


def make_plan(source=DEFAULT_SOURCE):
    originals = generate_full_plan()
    assert audit_full_plan(originals)["valid"]
    occupations = occupation_rows(source)
    l2s = {}
    for recipe in originals:
        labels = l2s.setdefault(recipe["domain"], [])
        if recipe["domain_l2"] not in labels:
            labels.append(recipe["domain_l2"])
    plan = []
    for index, original in enumerate(originals, 1):
        r = copy.deepcopy(original)
        code = MAPPING[r["domain"]][l2s[r["domain"]].index(r["domain_l2"])]
        occupation = occupations[code]  # Unknown official codes are hard failures.
        r["legacy_recipe_id"] = original["task_id"]
        r["task_id"] = f"ONET-LEGACY-{index:04d}"
        r["name"] = "work/" + r["task_id"].lower()
        r["seed"] = 20261020000 + index
        r["validation_profile"] = "legacy-reproduction"
        r["onet_mapping"] = {
            "database": "O*NET", "version": "31.0", "occupation_code": code,
            "occupation": occupation, "mapping_level": "authored_L2_correspondence",
            "is_official_domain_mapping": False, "task_id": None,
            "scope": "occupation_background_not_equivalent_professional_workflow",
            "occupation_url": f"https://www.onetonline.org/link/summary/{code}",
        }
        r["scenario"] += (
            f"\n\n职业分类参考：O*NET 31.0 {code} {occupation['Title']}。"
            f"职业定义：{occupation['Description']}"
        )
        assert all(r[k] == original[k] for k in PRESERVED)
        plan.append(r)
    return plan


def inspect(root):
    manifest = json.loads((root/"generation_manifest.json").read_text())
    recipe = manifest["recipe"]
    package = json.loads((root/"package_validation.json").read_text())
    golden = json.loads((root/"golden_validation.json").read_text())
    assert package.get("legacy_reproduction_complete") is True, root
    assert golden["passed"] and golden["score_final"] > .85, root
    assert all(v for k,v in package["full_builder_checks"].items() if k != "category_contract_valid"), root
    instruction = (root/"instruction.md").read_text()
    assert all(s in instruction for s in ["## 工作要求", "## 交付物", "## 硬约束与禁止项", "domain-audit-sop"])
    # Check every material/Skill/reference against the original builder's hashes.
    for name, digest in manifest["file_sha256"].items():
        assert hashlib.sha256((root/name).read_bytes()).hexdigest() == digest, (root,name)
    return {"id":root.name, "domain":recipe["domain"], "category":recipe["category"],
            "complexity":recipe["task_complexity"], "prompt_chars":len(instruction),
            "metrics":manifest["metrics"], "occupation_code":recipe["onet_mapping"]["occupation_code"],
            "strict_category_valid":package["category_contract"]["valid"],
            "strict_category_checks":package["category_contract"],
            "gold_score":golden["score_final"]}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--out", required=True, type=Path)
    parser.add_argument("--jobs", type=int, default=32)
    parser.add_argument("--smoke", action="store_true")
    args = parser.parse_args()
    plan = make_plan()
    if args.smoke:
        # Cover every domain x category, all C1-C5 and all fourteen weaknesses.
        selected, seen = [], set()
        for r in plan:
            keys = {(r["domain"],r["category"]), ("C",r["task_complexity"])}
            keys.update(("W",w) for w in r["weakness_tags"])
            if keys - seen:
                selected.append(r); seen.update(keys)
        plan = selected
    args.out.mkdir(parents=True, exist_ok=True)
    write_json(args.out/"plan.json", {"recipes":plan, "profile":"legacy-reproduction"})
    print(f"Building {len(plan)} recipes with original full_builder, jobs={args.jobs}", flush=True)
    result = build_all(plan,args.out/"tasks",jobs=args.jobs,resume=True)
    print(json.dumps({k:result[k] for k in ['total','built','resumed','failed']},ensure_ascii=False),flush=True)
    if result["failed"]:
        for r in result["results"]:
            if r["status"] == "failed": print(r["task_id"],r["error"][:1500],flush=True)
        return 1
    with ThreadPoolExecutor(max_workers=args.jobs) as pool:
        checks = list(pool.map(inspect,[args.out/"tasks"/r["task_id"] for r in plan]))
    report = {"tasks_materialized":len(checks), "builder":"workfactory.full_builder (existing)",
              "planner":"workfactory.full_plan (existing)", "model_calls":0,
              "domain_counts":dict(Counter(r['domain'] for r in checks)),
              "category_counts":dict(Counter(r['category'] for r in checks)),
              "complexity_counts":dict(Counter(r['complexity'] for r in checks)),
              "mapped_occupations":len({r['occupation_code'] for r in checks}),
              "local_gold_passed":len(checks),
              "strict_category_passed":sum(r['strict_category_valid'] for r in checks),
              "rewardkit_verified":False,"avg_prompt_chars":round(sum(r['prompt_chars'] for r in checks)/len(checks)),
              "note":"Historical full-template reproduction; strict category gaps retained and disclosed, not certified."}
    write_json(args.out/"audit_summary.json",report)
    write_json(args.out/"task_audit.json",checks)
    write_json(args.out/"builder_source_sha256.json", {
        str(p.relative_to(LEGACY)): hashlib.sha256(p.read_bytes()).hexdigest()
        for p in sorted((LEGACY/"workfactory").rglob("*.py"))})
    print(json.dumps(report,ensure_ascii=False),flush=True)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
