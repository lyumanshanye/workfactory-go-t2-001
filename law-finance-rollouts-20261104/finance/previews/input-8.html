531proc newe ( h , q ) ;
532
533 local u, t, i , s, c ;
534
535 t=rows(h) ;
536
537 u=h-(cen_nw.==1)* meanc(h)' ;
538 s=u'u / t ;
539
540 i=1 ; do while i<=q ;
541 c=u[i+1:t,.]'u[1:t-i,.] / t ;
542 s=s+ (1-(i/(q+1))) * ( c+c' ) ;
543 i=i+1 ; endo ;
544
545 retp( s ) ;
546endp;

803proc ols_r2 ( mom ) ;
804
805 local alpha1,res1,pre1, c,fu, alpha2,res2,pre2, kma,b ;
806
807 { alpha1,res1,pre1 } = OLSQR2(nw_sri'ones(t,1),nw_sr*mom) ;
808
809 c=pre1'pre1/(t^2) ;
810
811 fu=c;
812
813 if cen_nw==1 ;
814
815 { alpha2,res2,pre2 } = OLSQR2(nw_sr*ones(t,1),nw_sr*mom) ;
816
817 kma=res2'res2 ;
818 b=-pre1'pre2/t ;
819
820 fu=c/(kma*c+(1+b)^2) ;
821
822 endif;
823
824 retp( fu ) ;
825
826endp;
