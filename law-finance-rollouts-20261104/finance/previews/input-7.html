567proc newe ( h , q ) ;
568
569 local u, t, i , s, c ;
570
571 t=rows(h) ;
572
573 u=h-(cen_nw.==1)* meanc(h)' ;
574 s=u'u / t ;
575
576 i=1 ; do while i<=q ;
577 c=u[i+1:t,.]'u[1:t-i,.] / t ;

579 i=i+1 ; endo ;
