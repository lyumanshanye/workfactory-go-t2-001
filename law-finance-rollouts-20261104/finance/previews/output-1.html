88def objective(v,h,weight=None):

100def optimize(h,start,w=None,cu=False,multi=False):
101 starts=[start]
102 if multi:
103 rng=np.random.default_rng(202503)
104 starts += [start+rng.normal(size=len(start))*0.10 for _ in range(2)]
105 choices=[]
106 for x in starts:
107 if cu:
108 sol=minimize(lambda v:objective(v,h),x,method='BFGS',options={'maxiter':250,'gtol':2e-6})
109 else:
110 # equivalent minimization of mean moments' quadratic norm at fixed weighting matrix
111 try: root=np.linalg.cholesky(w).T
112 except np.linalg.LinAlgError: root=np.linalg.cholesky(w+np.eye(len(w))*1e-8).T
113 sol=least_squares(lambda v:root@h(v).mean(axis=0),x,max_nfev=180,ftol=1e-10,xtol=1e-10,gtol=1e-10)
114 choices.append((objective(sol.x,h,None if cu else w),sol))
115 choices.sort(key=lambda pair:pair[0]);return choices[0][1], [v for v,_ in choices]
